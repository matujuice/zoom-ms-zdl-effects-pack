// Zoom MS bar sync v3, for Mozaic (AUM). Keeps our effects on the bar.
// Needs the twin-Tempo builds (DualShft, Choral, EuGate, DubSiren, Scrub with
// Tempo 0..441): screen T = BPM T, screen T + 201 = the same BPM again, and
// flipping between the two restarts the LFO / pattern / grain.
// AUM routing: Mozaic MIDI out -> Zoom, Zoom MIDI out -> Mozaic in. AUM is the
// master clock; start AUM's transport.
//
// Every N bars, Early ms before the downbeat, the script flips the Tempo knob of
// ONE effect, in the slot the Send knob picks (1..3), to its other copy. When the
// host tempo changes it sends the new BPM to that knob and to the patch tempo.
// That effect passes the bar on to every later slot; set the other synced effects'
// Tempo to FOLLW (v3, 2026-10-07: one sender is all a patch needs).
// Tempo is the 8th knob on every synced effect, so the script does not need to know
// which effect is in the Send slot. Only the first 3 effects of a patch accept edits.
//
// Knobs: 0 Early (0..127 ms; default 17: 12 measured on the MS-60B + iPhone 2026-10-05, plus 5)
//        1 Every (1, 2, 4 or 8 bars)
//        2 Tempo (100..227 BPM, used when Host tempo is off)
//        3 Send: OFF / Slot 1 / Slot 2 / Slot 3 (the one slot that gets the flips)
// Pads (Mozaic shows 4): 0 Sync on/off   1 Host tempo on/off   2 Sync now
//        3 pedal: MS-50G (ID 58, also an MS-60B on MS-50G firmware) / MS-60B (ID 5F)
// (Edit enable stays in the code as hidden pad 7.)
// Log shows what was sent.

@OnLoad
  zid = 0x58
  early = 17
  every = 1
  manual = 160
  tempo = 160
  lastsent = 0
  follow = 1
  syncon = 1
  twin = 0
  armed = 0
  fireat = 0
  send = 1
  // Mozaic wants every variable set here before other events use it
  barlen = 0
  nextbar = 0
  bar = 0
  ps = 0
  pp = 0
  pv = 0
  k = 0
  v = 0
  e = 0
  t = 0
  p = 0
  SetKnobValue 0, early
  SetKnobValue 1, 0
  SetKnobValue 2, manual - 100
  SetKnobValue 3, 48
  LabelKnobs {Zoom bar sync}
  LabelKnob 0, {Early }, early, {ms}
  LabelKnob 1, {Every }, every, { bar}
  LabelKnob 2, {BPM }, manual
  Call @LabelSend
  LabelPad 0, {Sync ON}
  LabelPad 1, {Host tempo ON}
  LabelPad 2, {Sync now}
  LabelPad 3, {MS-50G}
  LabelPad 7, {Edit enable}
  SetTimerInterval 5
  StartTimer
  Call @SendEnable
  Log {Loaded. Send = slot of the effect that keeps the bar, others on FOLLW. Start AUM.}
@End

@LabelSend
  if send = 0
    LabelKnob 3, {Send OFF}
  else
    LabelKnob 3, {Send slot }, send
  endif
@End

@SendEnable
  syx[0] = 0x52
  syx[1] = 0x00
  syx[2] = zid
  syx[3] = 0x50
  SendSysex syx, 4
@End

// Sends ps / pp / pv as a parameter edit
@SendParam
  syx[0] = 0x52
  syx[1] = 0x00
  syx[2] = zid
  syx[3] = 0x31
  syx[4] = ps
  syx[5] = pp
  syx[6] = pv % 128
  syx[7] = Div pv, 128
  SendSysex syx, 8
@End

// Sends the Tempo knob of the Send slot, on the current copy (twin)
@SendKnobs
  if send > 0
    ps = send - 1
    // Tempo is the 8th knob on every effect (param 2 = knob 1)
    pp = 9
    pv = tempo + 201 * twin
    Call @SendParam
  endif
@End

@SendTempo
  if tempo < 40
    tempo = 40
  elseif tempo > 240
    tempo = 240
  endif
  ps = 3
  pp = 8
  pv = tempo
  Call @SendParam
  Call @SendKnobs
  lastsent = tempo
  Log {Tempo }, tempo
@End

@Flip
  twin = 1 - twin
  Call @SendKnobs
@End

@OnKnobChange
  k = LastKnob
  v = GetKnobValue k
  if k = 0
    early = Round v
    LabelKnob 0, {Early }, early, {ms}
  elseif k = 1
    e = RoundDown (v / 32)
    if e = 0
      every = 1
    elseif e = 1
      every = 2
    elseif e = 2
      every = 4
    else
      every = 8
    endif
    LabelKnob 1, {Every }, every, { bar}
  elseif k = 2
    manual = 100 + Round v
    LabelKnob 2, {BPM }, manual
    if follow = 0
      tempo = manual
      Call @SendTempo
    endif
  elseif k = 3
    e = RoundDown (v / 32)
    if e > 3
      e = 3
    endif
    if e <> send
      send = e
      Call @LabelSend
      Call @SendEnable
      Call @SendKnobs
    endif
  endif
@End

@OnPadDown
  p = LastPad
  if p = 0
    syncon = 1 - syncon
    armed = 0
    if syncon = 1
      LabelPad 0, {Sync ON}
    else
      LabelPad 0, {Sync OFF}
    endif
  elseif p = 1
    follow = 1 - follow
    if follow = 1
      LabelPad 1, {Host tempo ON}
      tempo = Round HostTempo
    else
      LabelPad 1, {Host tempo OFF}
      tempo = manual
    endif
    Call @SendTempo
  elseif p = 2
    Call @Flip
    Log {Sync now}
  elseif p = 3
    if zid = 0x58
      zid = 0x5F
      LabelPad 3, {MS-60B}
    else
      zid = 0x58
      LabelPad 3, {MS-50G}
    endif
    Call @SendEnable
  elseif p = 7
    Call @SendEnable
    Log {Edit enable sent}
  endif
@End

// Stops Mozaic passing other MIDI straight through
@OnMidiInput
@End

@OnHostStart
  Call @SendEnable
  armed = 0
  if follow = 1
    tempo = Round HostTempo
  endif
  Call @SendTempo
  // The first downbeat can't be sent early: restart now
  if syncon = 1
    Call @Flip
  endif
  Log {Start}
@End

@OnNewBar
  barlen = 240000 / HostTempo
  nextbar = SystemTime + barlen
  bar = RoundDown HostBar
  if follow = 1
    t = Round HostTempo
    if t <> lastsent
      tempo = t
      Call @SendTempo
    endif
  endif
  // The bar event can arrive just before the timer reaches fireat: then the flip is due now
  if armed = 1
    armed = 0
    Call @Flip
  endif
  // Arm the flip for the next downbeat if that bar starts a group of Every bars
  // (host bars 0, Every, 2 x Every ...)
  if syncon = 1 and (bar + 1) % every = 0
    fireat = nextbar - early
    armed = 1
  endif
@End

@OnTimer
  if armed = 1 and SystemTime >= fireat
    armed = 0
    Call @Flip
  endif
@End
